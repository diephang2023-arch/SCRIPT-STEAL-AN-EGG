local function applyInstantHold(prompt)
    if prompt and prompt:IsA("ProximityPrompt") then
        pcall(function()
            prompt.HoldDuration = 0
        end)
    end
end

for _, desc in ipairs(Workspace:GetDescendants()) do
    if desc:IsA("ProximityPrompt") then
        applyInstantHold(desc)
    end
end

Workspace.DescendantAdded:Connect(function(desc)
    if desc:IsA("ProximityPrompt") then
        task.defer(applyInstantHold, desc)
    end
end)

local fastGrabBusy = false
ProximityPromptService.PromptButtonHoldBegan:Connect(function(prompt, player)
    if player and player ~= LocalPlayer then return end
    if fastGrabBusy or not prompt or not prompt.Enabled then return end
    fastGrabBusy = true

    pcall(function()
        prompt.HoldDuration = 0
        if fireproximityprompt then
            fireproximityprompt(prompt)
        else
            prompt:InputHoldBegin()
            task.wait(0.02)
            prompt:InputHoldEnd()
        end
    end)

    task.delay(0.05, function()
        fastGrabBusy = false
    end)
end)

-- PHẦN THÊM 2: CÓ TRỨNG LÀ MANG VỀ NHÀ NGAY KHI BẬT CÔNG TẮC
-- ============================================================
local HOME_POSITION = Vector3.new(0, 5, 0)
local RETURN_DELAY = 0.05
local RETURN_COOLDOWN = 1.5
local lastReturnTime = 0
local isReturningHome = false

local function hasEggInHandOrBag()
    local char = LocalPlayer.Character
    if char then
        for _, obj in ipairs(char:GetChildren()) do
            local nameLower = string.lower(obj.Name)
            if (obj:IsA("Tool") or obj:IsA("Model") or obj:IsA("BasePart")) and string.find(nameLower, "egg", 1, true) then
                return true
            end
        end
        for _, attrVal in pairs(char:GetAttributes()) do
            if type(attrVal) == "string" and string.find(string.lower(attrVal), "egg", 1, true) then
                return true
            end
        end
    end

    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if string.find(string.lower(item.Name), "egg", 1, true) then
                return true
            end
        end
    end

    return false
end

local function bringEggHomeNow()
    if isReturningHome then return end
    if os.clock() - lastReturnTime < RETURN_COOLDOWN then return end

    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not char or not root then return end

    isReturningHome = true
    lastReturnTime = os.clock()

    pcall(function()
        task.wait(RETURN_DELAY)
        if State.AntiGuard and root.Parent then
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
            root.CFrame = CFrame.new(HOME_POSITION + Vector3.new(0, 2, 0))
        end
    end)

    task.delay(0.3, function()
        isReturningHome = false
    end)
end

task.spawn(function()
    while true do
        if State.AntiGuard and not isExecuting and not isReturningHome then
            if hasEggInHandOrBag() then
                bringEggHomeNow()
            end
        end
        task.wait(0.08)
    end
end)

-- PHẦN THÊM 3: NHẮM VÀO BẪY KHI CẦM TRỨNG (GIỮ AN TOÀN MAP PET)
-- ============================================================
local TRAP_KEYWORDS = {
    "trap", "bẫy", "bay", "spike", "mine", "snare",
    "laser", "shock", "net", "cage", "bear", "hazard"
}

local WALL_KEYWORDS = {
    "wall", "tường", "tuong", "floor", "sàn", "san",
    "ground", "pet", "pen", "fence", "gate", "door", "building", "house"
}

local neutralizedCache = {}
local lastEggRef = nil

local function isWallOrPetMapPart(obj)
    if not obj then return false end
    local cur = obj
    for _ = 1, 6 do
        if not cur or cur == Workspace then break end
        local n = string.lower(tostring(cur.Name or ""))
        for _, kw in ipairs(WALL_KEYWORDS) do
            if string.find(n, kw, 1, true) then
                return true
            end
        end
        cur = cur.Parent
    end
    return false
end

local function getHeldEggTool()
    local char = LocalPlayer.Character
    if not char then return nil end
    for _, child in ipairs(char:GetChildren()) do
        if child:IsA("Tool") and string.find(string.lower(child.Name), "egg", 1, true) then
            return child
        end
    end
    return nil
end

local function findTrapTarget(inst)
    if not inst or isWallOrPetMapPart(inst) then return nil end
    local cur = inst
    for _ = 1, 5 do
        if not cur or cur == Workspace then break end
        local n = string.lower(tostring(cur.Name or ""))
        for _, kw in ipairs(TRAP_KEYWORDS) do
            if string.find(n, kw, 1, true) then
                return cur
            end
        end
        cur = cur.Parent
    end
    return nil
end

local function getAimedObject()
    if Mouse and Mouse.Target then
        return Mouse.Target
    end

    local cam = Workspace.CurrentCamera
    if not cam then return nil end

    local vp = cam.ViewportSize
    local ray = cam:ViewportPointToRay(vp.X * 0.5, vp.Y * 0.5)

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    if LocalPlayer.Character then
        params.FilterDescendantsInstances = { LocalPlayer.Character }
    end

    local hit = Workspace:Raycast(ray.Origin, ray.Direction * 150, params)
    return hit and hit.Instance or nil
end

local function neutralizeTrapPart(obj)
    if not obj or isWallOrPetMapPart(obj) then return end
    pcall(function()
        if obj:IsA("BasePart") then
            pcall(function() obj.CanTouch = false end)
            for _, child in ipairs(obj:GetChildren()) do
                if child:IsA("TouchTransmitter") then
                    child:Destroy()
                end
            end
        elseif obj:IsA("TouchTransmitter") then
            obj:Destroy()
        elseif obj:IsA("ProximityPrompt") then
            obj.Enabled = false
        end
    end)
end

RunService.Heartbeat:Connect(function()
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not char or not hum then return end

    local heldEgg = getHeldEggTool()
    if heldEgg then
        lastEggRef = heldEgg
    end

    local aimed = getAimedObject()
    if not aimed then return end

    local trapObj = findTrapTarget(aimed)
    if trapObj then
        if not neutralizedCache[trapObj] then
            neutralizedCache[trapObj] = true
            for _, d in ipairs(trapObj:GetDescendants()) do
                neutralizeTrapPart(d)
            end
            neutralizeTrapPart(trapObj)
        end

        if lastEggRef and lastEggRef.Parent == LocalPlayer:FindFirstChild("Backpack") then
            pcall(function()
                hum:EquipTool(lastEggRef)
            end)
        end
    end
end)

local function hasObstacleAhead(char, origin, dirUnit, distance)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { char }

    local hit = Workspace:Raycast(origin, dirUnit * distance, params)
    if hit and hit.Instance and hit.Instance.CanCollide then
        return true
    end
    return false
end

RunService.Heartbeat:Connect(function()
    if not SpeedConfig.Enabled then return end

    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")

    if not char or not hum or not root or hum.Health <= 0 then return end

    if hum.WalkSpeed ~= SpeedConfig.CurrentSpeed then
        hum.WalkSpeed = SpeedConfig.CurrentSpeed
    end

    local moveDir = hum.MoveDirection
    if moveDir.Magnitude > 0.05 and SpeedConfig.CurrentSpeed > 16 then
        local flatDir = Vector3.new(moveDir.X, 0, moveDir.Z)
        if flatDir.Magnitude > 0.01 then
            flatDir = flatDir.Unit
            local blocked = hasObstacleAhead(char, root.Position, flatDir, 3.2)

            if not blocked then
                local currentVel = root.AssemblyLinearVelocity
                local desiredVel = flatDir * SpeedConfig.CurrentSpeed
                local blendedVel = currentVel:Lerp(Vector3.new(desiredVel.X, currentVel.Y, desiredVel.Z), 0.35)
                root.AssemblyLinearVelocity = blendedVel
            end
        end
    end
end)

    local hum = newChar:WaitForChild("Humanoid", 5)
    if hum and SpeedConfig.Enabled then
        task.wait(0.1)
        hum.WalkSpeed = SpeedConfig.CurrentSpeed
    end
end)

local function restoreCharacterCollisions()
    for part, canCollide in pairs(savedCollideParts) do
        if part and part.Parent then
            pcall(function()
                part.CanCollide = canCollide
            end)
        end
    end
    savedCollideParts = {}

    local char = LocalPlayer.Character
    if char then
        for _, desc in ipairs(char:GetDescendants()) do
            if desc:IsA("BasePart") and (desc.Name == "HumanoidRootPart" or desc.Name == "Torso" or desc.Name == "UpperTorso" or desc.Name == "LowerTorso") then
                pcall(function()
                    desc.CanCollide = true
                end)
            end
        end
    end
end
