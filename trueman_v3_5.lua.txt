local function tween(obj, t, props)
    local tw = TS:Create(obj, TweenInfo.new(t, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), props)

    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        oDragging = true
        oDragStart = input.Position
        oStartPos = oBtn.Position
        oMoved = 0
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                oDragging = false
            end
        end)
    end
end)

oBtn.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        oDragInput = input
    end
end)

UIS.InputChanged:Connect(function(input)
    if input == oDragInput and oDragging then
        local d = input.Position - oDragStart
        oMoved = d.Magnitude
        oBtn.Position = UDim2.new(oStartPos.X.Scale, oStartPos.X.Offset + d.X, oStartPos.Y.Scale, oStartPos.Y.Offset + d.Y)
    end
end)

local function switchCat(name)
    curCat = name
    for pName, page in pairs(pages) do
        page.Visible = (pName == name)
    end
    for bName, b in pairs(catBtns) do
        tween(b, 0.18, {BackgroundColor3 = (bName == name) and COL.pink or COL.panel})
    end
end

    btn.Size = UDim2.fromScale(1, 1)
    btn.BackgroundTransparency = 1
    btn.Text = ""
    btn.ZIndex = 16

    local function update(x)
        local rel = math.clamp((x - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
        local v = math.floor(minV + rel * (maxV - minV))
        fill.Size = UDim2.new(rel, 0, 1, 0)
        valLbl.Text = tostring(v) .. " studs/s"
        if cb then cb(v) end
    end

    btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            drag = true
            update(input.Position.X)
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if drag and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            update(input.Position.X)
        end
    end)

    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            drag = false
        end
    end)
    return row
end

            VU:Button2Up(Vector2.zero, workspace.CurrentCamera.CFrame)
        end)
    end

local hbConn = RS.Heartbeat:Connect(function()
    local char = P.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")

    if hum then
        if F.Speed then
            if hum.WalkSpeed ~= F.SpeedVal then
                hum.WalkSpeed = F.SpeedVal
            end
            if root and hum.MoveDirection.Magnitude > 0 then
                root.CFrame = root.CFrame + (hum.MoveDirection * (F.SpeedVal / 60))
            end
        end

        if F.GodMode and hum.Health > 0 and hum.Health < hum.MaxHealth then
            hum.Health = hum.MaxHealth
        end

        if F.AntiRagdoll then
            pcall(function()
                hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
                hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
            end)
        end
    end
end)
