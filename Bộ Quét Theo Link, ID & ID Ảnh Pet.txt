local SELF_SIGNATURE = "--[[ GAU_TARGET_LINK_SCANNER_SELF_IGNORE ]]"

local targetExecuted = false
local targetHosts = {}

local function normalizeUrl(u)
	if type(u) ~= "string" then
		return ""
	end
	return u:gsub("^%s+", ""):gsub("%s+$", "")
end

local function extractHost(u)
	local host = tostring(u or ""):match("^https?://([^/%?#]+)")
	return host and host:lower() or ""
end

do
	local mainHost = extractHost(TARGET_SCRIPT_URL)
	if mainHost ~= "" then
		targetHosts[mainHost] = true
	end
end

local function isTargetRelatedUrl(u)
	local clean = normalizeUrl(u)
	if clean == "" then
		return false
	end
	local targetClean = normalizeUrl(TARGET_SCRIPT_URL)
	if targetClean ~= "" and (clean == targetClean or clean:find(targetClean, 1, true) or targetClean:find(clean, 1, true)) then
		return true
	end
	local h = extractHost(clean)
	if h ~= "" and targetHosts[h] then
		return true
	end
	return targetExecuted
end

local CAPTURED = {
	TEXTS = {},
	IMAGES = {},
	IDS = {},
	DISCORD = {},
	URLS = {},
	CLIPBOARD = {},
}

local SEEN = {
	TEXTS = {},
	IMAGES = {},
	IDS = {},
	DISCORD = {},
	URLS = {},
	CLIPBOARD = {},
	SOURCES = {},
}

local onCountUpdated = nil

local function isSelfContent(str)
	if type(str) ~= "string" then
		return false
	end
	return string.find(str, SELF_SIGNATURE, 1, true) ~= nil
		or (string.find(str, "GauTargetScannerGui", 1, true) ~= nil and string.find(str, "TARGET_SCRIPT_URL", 1, true) ~= nil)
end

local function totalCapturedCount()
	return #CAPTURED.TEXTS + #CAPTURED.IMAGES + #CAPTURED.IDS + #CAPTURED.DISCORD + #CAPTURED.URLS + #CAPTURED.CLIPBOARD
end

local function notifyCountChanged()
	if type(onCountUpdated) == "function" then
		pcall(onCountUpdated, totalCapturedCount())
	end
end

local function addUniqueItem(category, value, sourceTag)
	if not targetExecuted then
		return
	end
	if type(value) ~= "string" then
		return
	end
	local trimmed = value:gsub("^%s+", ""):gsub("%s+$", "")
	if trimmed == "" or #trimmed > 500 then
		return
	end
	if isSelfContent(trimmed) then
		return
	end
	if SEEN[category][trimmed] then
		return
	end
	SEEN[category][trimmed] = true

	table.insert(CAPTURED[category], {
		value = trimmed,
		source = sourceTag or "TargetExecute",
	})
	notifyCountChanged()
end

local function extractAndCaptureIds(rawStr, sourceTag)
	if not targetExecuted or type(rawStr) ~= "string" or rawStr == "" then
		return
	end

	-- 1. Bắt toàn bộ chuỗi rbxassetid://, rbxthumb://, rbxgameasset://... và số ID bên trong
	for rbxUri in rawStr:gmatch("rbx[a-z]+://[%w%d_%-%./%?&=]+") do
		addUniqueItem("IMAGES", rbxUri, sourceTag)
		addUniqueItem("IDS", rbxUri, sourceTag)
		for idNum in rbxUri:gmatch("(%d+)") do
			if #idNum >= 4 then
				addUniqueItem("IDS", idNum, sourceTag)
			end
		end
	end

	-- 2. Bắt các trường / từ khóa liên quan tới ID, Pet, Image, Icon, Asset, Texture
	for keyId in rawStr:gmatch('[%w_]*[Ii][Dd][%w_]*%s*[:=]?%s*["\']?([%w%d_:/%-%.]+)["\']?') do
		if #keyId >= 4 then
			addUniqueItem("IDS", keyId, sourceTag)
		end
	end

	for petImg in rawStr:gmatch('[Pp][Ee][Tt][%w_]*%s*[:=]?%s*["\']?([%w%d_:/%-%.]+)["\']?') do
		if #petImg >= 4 then
			addUniqueItem("IDS", petImg, sourceTag .. ":Pet")
		end
	end

	-- 3. Bắt số ID thuần túy (từ 4 chữ số trở lên) nếu nằm trong chuỗi/text/code
	for numToken in rawStr:gmatch("%f[%d]%d%d%d+%f[^%d]") do
		if #numToken >= 4 and #numToken <= 20 then
			addUniqueItem("IDS", numToken, sourceTag)
		end
	end
end

-- 3. ÁP DỤNG LỚP PHỦ THAY THẾ (CHỈ CHẠY KHI ENABLE_CUSTOM_OVERLAY = true)
-- =====================================================================
local function applyStringOverlay(rawStr)
	if not ENABLE_CUSTOM_OVERLAY or type(rawStr) ~= "string" or rawStr == "" then
		return rawStr
	end

	if OVERLAY_CONFIG.TEXT[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.TEXT[rawStr])
	end
	if OVERLAY_CONFIG.IDS[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.IDS[rawStr])
	end
	if OVERLAY_CONFIG.LINKS[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.LINKS[rawStr])
	end
	if OVERLAY_CONFIG.IMAGES[rawStr] ~= nil then
		return tostring(OVERLAY_CONFIG.IMAGES[rawStr])
	end

	local out = rawStr
	for oldId, newId in pairs(OVERLAY_CONFIG.IDS) do
		if type(oldId) == "string" and oldId ~= "" and string.find(out, oldId, 1, true) then
			out = out:gsub(oldId:gsub("(%W)", "%%%1"), tostring(newId))
		end
	end
	for oldTxt, newTxt in pairs(OVERLAY_CONFIG.TEXT) do
		if type(oldTxt) == "string" and oldTxt ~= "" and string.find(out, oldTxt, 1, true) then
			out = out:gsub(oldTxt:gsub("(%W)", "%%%1"), tostring(newTxt))
		end
	end
	for oldUrl, newUrl in pairs(OVERLAY_CONFIG.LINKS) do
		if type(oldUrl) == "string" and oldUrl ~= "" and string.find(out, oldUrl, 1, true) then
			out = out:gsub(oldUrl:gsub("(%W)", "%%%1"), tostring(newUrl))
		end
	end
	return out
end

local function applyImageOverlay(rawImg)
	if not ENABLE_CUSTOM_OVERLAY or type(rawImg) ~= "string" or rawImg == "" then
		return rawImg
	end

	if OVERLAY_CONFIG.IMAGES[rawImg] ~= nil then
		return tostring(OVERLAY_CONFIG.IMAGES[rawImg])
	end
	if OVERLAY_CONFIG.IDS[rawImg] ~= nil then
		return tostring(OVERLAY_CONFIG.IDS[rawImg])
	end

	local digits = rawImg:match("(%d+)")
	if digits then
		if OVERLAY_CONFIG.IMAGES[digits] ~= nil then
			local rep = tostring(OVERLAY_CONFIG.IMAGES[digits])
			return rep:match("^rbx") and rep or ("rbxassetid://" .. rep)
		end
		if OVERLAY_CONFIG.IDS[digits] ~= nil then
			local rep = tostring(OVERLAY_CONFIG.IDS[digits])
			return rep:match("^rbx") and rep or ("rbxassetid://" .. rep)
		end
		local fullKey = "rbxassetid://" .. digits
		if OVERLAY_CONFIG.IMAGES[fullKey] ~= nil then
			local rep = tostring(OVERLAY_CONFIG.IMAGES[fullKey])
			return rep:match("^rbx") and rep or ("rbxassetid://" .. rep)
		end
		if OVERLAY_CONFIG.IDS[fullKey] ~= nil then
			local rep = tostring(OVERLAY_CONFIG.IDS[fullKey])
			return rep:match("^rbx") and rep or ("rbxassetid://" .. rep)
		end
	end

	return rawImg
end

local function applyClipboardOverlay(rawClip)
	if not ENABLE_CUSTOM_OVERLAY or type(rawClip) ~= "string" or rawClip == "" then
		return rawClip
	end

	if OVERLAY_CONFIG.CLIPBOARD[rawClip] ~= nil then
		return tostring(OVERLAY_CONFIG.CLIPBOARD[rawClip])
	end
	if OVERLAY_CONFIG.IDS[rawClip] ~= nil then
		return tostring(OVERLAY_CONFIG.IDS[rawClip])
	end
	if OVERLAY_CONFIG.LINKS[rawClip] ~= nil then
		return tostring(OVERLAY_CONFIG.LINKS[rawClip])
	end

	local out = rawClip
	for oldId, newId in pairs(OVERLAY_CONFIG.IDS) do
		if type(oldId) == "string" and oldId ~= "" and string.find(out, oldId, 1, true) then
			out = out:gsub(oldId:gsub("(%W)", "%%%1"), tostring(newId))
		end
	end
	for oldVal, newVal in pairs(OVERLAY_CONFIG.CLIPBOARD) do
		if type(oldVal) == "string" and oldVal ~= "" and string.find(out, oldVal, 1, true) then
			out = out:gsub(oldVal:gsub("(%W)", "%%%1"), tostring(newVal))
		end
	end
	for oldUrl, newUrl in pairs(OVERLAY_CONFIG.LINKS) do
		if type(oldUrl) == "string" and oldUrl ~= "" and string.find(out, oldUrl, 1, true) then
			out = out:gsub(oldUrl:gsub("(%W)", "%%%1"), tostring(newUrl))
		end
	end
	return out
end

-- 4. QUÉT NỘI DUNG TẢI VỀ & GIAO DIỆN UI SAU KHI EXECUTE ĐÚNG LINK
-- =====================================================================
local function scanFetchedText(rawStr, sourceTag)
	if not targetExecuted then
		return
	end
	if type(rawStr) ~= "string" or #rawStr == 0 or isSelfContent(rawStr) then
		return
	end

	local sig = tostring(#rawStr) .. ":" .. rawStr:sub(1, 60) .. ":" .. rawStr:sub(-60)
	if SEEN.SOURCES[sig] then
		return
	end
	SEEN.SOURCES[sig] = true

	for subUrl in rawStr:gmatch("https?://[%w%.%-_~:/%?#@!$&'%(%)%*%+,;=%%]+") do
		local subHost = extractHost(subUrl)
		if subHost ~= "" then
			targetHosts[subHost] = true
		end
	end

	for dLink in rawStr:gmatch("https?://[%w%.%-_]*discord[%w%.%-_/%%?&=]+") do
		addUniqueItem("DISCORD", dLink, sourceTag)
	end
	for dShort in rawStr:gmatch("discord%.gg/[%w%-_]+") do
		addUniqueItem("DISCORD", dShort, sourceTag)
	end

	for url in rawStr:gmatch("https?://[%w%.%-_~:/%?#@!$&'%(%)%*%+,;=%%]+") do
		if not url:lower():find("discord", 1, true) then
			addUniqueItem("URLS", url, sourceTag)
		end
	end

	extractAndCaptureIds(rawStr, sourceTag)

	for clipStr in rawStr:gmatch('setclipboard%s*%(%s*["\']([^"\'\r\n]+)["\']%s*%)') do
		addUniqueItem("CLIPBOARD", clipStr, sourceTag)
		extractAndCaptureIds(clipStr, sourceTag .. ":Clipboard")
	end
end

local function isOwnGuiInstance(inst)
	local curr = inst
	while curr do
		if curr.Name == "GauTargetScannerGui" then
			return true
		end
		curr = curr.Parent
	end
	return false
end

local hookedInstances = setmetatable({}, { __mode = "k" })

local function inspectAndHookGuiObject(obj)
	if not obj or isOwnGuiInstance(obj) then
		return
	end

	if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
		local txt = obj.Text
		if type(txt) == "string" and txt ~= "" and targetExecuted then
			if txt:lower():find("discord%.gg") or txt:lower():find("discord%.com") then
				addUniqueItem("DISCORD", txt, "UI:" .. obj.ClassName)
			elseif txt:match("^https?://") then
				addUniqueItem("URLS", txt, "UI:" .. obj.ClassName)
			else
				addUniqueItem("TEXTS", txt, "UI:" .. obj.ClassName)
			end
			extractAndCaptureIds(txt, "UI:" .. obj.ClassName)
		end

		if ENABLE_CUSTOM_OVERLAY and type(txt) == "string" and txt ~= "" then
			local replaced = applyStringOverlay(txt)
			if replaced ~= txt then
				obj.Text = replaced
			end
		end

		if obj:IsA("TextBox") and obj.PlaceholderText ~= "" then
			if targetExecuted then
				addUniqueItem("TEXTS", obj.PlaceholderText, "UI:Placeholder")
				extractAndCaptureIds(obj.PlaceholderText, "UI:Placeholder")
			end
			if ENABLE_CUSTOM_OVERLAY then
				local repPh = applyStringOverlay(obj.PlaceholderText)
				if repPh ~= obj.PlaceholderText then
					obj.PlaceholderText = repPh
				end
			end
		end

		if not hookedInstances[obj] then
			hookedInstances[obj] = true
			obj:GetPropertyChangedSignal("Text"):Connect(function()
				local newTxt = obj.Text
				if type(newTxt) == "string" and newTxt ~= "" then
					if targetExecuted then
						if newTxt:lower():find("discord%.gg") or newTxt:lower():find("discord%.com") then
							addUniqueItem("DISCORD", newTxt, "UI:" .. obj.ClassName)
						elseif newTxt:match("^https?://") then
							addUniqueItem("URLS", newTxt, "UI:" .. obj.ClassName)
						else
							addUniqueItem("TEXTS", newTxt, "UI:" .. obj.ClassName)
						end
						extractAndCaptureIds(newTxt, "UI:" .. obj.ClassName)
					end

					if ENABLE_CUSTOM_OVERLAY then
						local mapped = applyStringOverlay(newTxt)
						if mapped ~= newTxt then
							obj.Text = mapped
						end
					end
				end
			end)
		end
	end

	if obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
		local img = obj.Image
		if type(img) == "string" and img ~= "" then
			if targetExecuted then
				addUniqueItem("IMAGES", img, "UI:" .. obj.ClassName)
				extractAndCaptureIds(img, "UI:" .. obj.ClassName)
			end
			if ENABLE_CUSTOM_OVERLAY then
				local repImg = applyImageOverlay(img)
				if repImg ~= img then
					obj.Image = repImg
				end
			end
		end

		if not hookedInstances[obj] then
			hookedInstances[obj] = true
			obj:GetPropertyChangedSignal("Image"):Connect(function()
				local newImg = obj.Image
				if type(newImg) == "string" and newImg ~= "" then
					if targetExecuted then
						addUniqueItem("IMAGES", newImg, "UI:" .. obj.ClassName)
						extractAndCaptureIds(newImg, "UI:" .. obj.ClassName)
					end
					if ENABLE_CUSTOM_OVERLAY then
						local mappedImg = applyImageOverlay(newImg)
						if mappedImg ~= newImg then
							obj.Image = mappedImg
						end
					end
				end
			end)
		end
	end
end

local function attachGuiWatcher(rootContainer)
	if not rootContainer then
		return
	end
	pcall(function()
		for _, desc in ipairs(rootContainer:GetDescendants()) do
			inspectAndHookGuiObject(desc)
		end
		rootContainer.DescendantAdded:Connect(function(desc)
			task.defer(inspectAndHookGuiObject, desc)
		end)
	end)
end

if gethui then
	pcall(function()
		attachGuiWatcher(gethui())
	end)
end
attachGuiWatcher(CoreGui)
if not PlayerGui and LocalPlayer then
	PlayerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui")
end
if PlayerGui then
	attachGuiWatcher(PlayerGui)
end

-- =====================================================================
-- 5. HOOK HTTPGET & SETCLIPBOARD (CHỈ TÍNH KHI EXECUTE ĐÚNG LINK)
-- =====================================================================
if hookfunction then
	if game.HttpGet then
		pcall(function()
			local oldHttpGet
			oldHttpGet = hookfunction(game.HttpGet, function(self, url, ...)
				local isMatch = isTargetRelatedUrl(url)
				if isMatch then
					targetExecuted = true
				end

				local res = oldHttpGet(self, url, ...)

				if isMatch and type(url) == "string" then
					if url:lower():find("discord") then
						addUniqueItem("DISCORD", url, "HttpGet:URL")
					else
						addUniqueItem("URLS", url, "HttpGet:URL")
					end
					extractAndCaptureIds(url, "HttpGet:URL")
				end

				if isMatch and type(res) == "string" and #res > 0 and not isSelfContent(res) then
					scanFetchedText(res, "HttpGet:" .. tostring(url))
				end

				return res
			end)
		end)
	end

	local clipTarget = setclipboard or toclipboard
	if type(clipTarget) == "function" then
		pcall(function()
			local oldClip
			oldClip = hookfunction(clipTarget, function(text, ...)
				if type(text) == "string" and text ~= "" and not isSelfContent(text) then
					if targetExecuted then
						if text:lower():find("discord%.gg") or text:lower():find("discord%.com") then
							addUniqueItem("DISCORD", text, "ClipboardButton")
						end
						addUniqueItem("CLIPBOARD", text, "ClipboardButton")
						extractAndCaptureIds(text, "ClipboardButton")
					end
					if ENABLE_CUSTOM_OVERLAY then
						text = applyClipboardOverlay(text)
					end
				end
				return oldClip(text, ...)
			end)
		end)
	end
end

local function escapeLuaStr(s)
	return tostring(s or ""):gsub("\\", "\\\\"):gsub('"', '\\"'):gsub("\r", ""):gsub("\n", "\\n")
end

local function formatEntries(list, mapTable)
	local out = {}
	for _, item in ipairs(list) do
		local chosen = (mapTable and mapTable[item.value] and mapTable[item.value] ~= "") and mapTable[item.value] or item.value
		table.insert(out, string.format('\t\t["%s"] = "%s",', escapeLuaStr(item.value), escapeLuaStr(chosen)))
	end
	return table.concat(out, "\n")
end

		"\t\tif OVERLAY_CONFIG.IMAGES[digits] ~= nil then",
		"\t\t\tlocal rep = tostring(OVERLAY_CONFIG.IMAGES[digits])",
		"\t\t\treturn rep:match(\"^rbx\") and rep or (\"rbxassetid://\" .. rep)",
		"\t\tend",
		"\t\tif OVERLAY_CONFIG.IDS[digits] ~= nil then",
		"\t\t\tlocal rep = tostring(OVERLAY_CONFIG.IDS[digits])",
		"\t\t\treturn rep:match(\"^rbx\") and rep or (\"rbxassetid://\" .. rep)",
		"\t\tend",
		"\t\tlocal fullKey = \"rbxassetid://\" .. digits",
		"\t\tif OVERLAY_CONFIG.IMAGES[fullKey] ~= nil then",
		"\t\t\tlocal rep = tostring(OVERLAY_CONFIG.IMAGES[fullKey])",
		"\t\t\treturn rep:match(\"^rbx\") and rep or (\"rbxassetid://\" .. rep)",
		"\t\tend",
		"\t\tif OVERLAY_CONFIG.IDS[fullKey] ~= nil then",
		"\t\t\tlocal rep = tostring(OVERLAY_CONFIG.IDS[fullKey])",
		"\t\t\treturn rep:match(\"^rbx\") and rep or (\"rbxassetid://\" .. rep)",
		"\t\tend",
		"\tend",
		"\treturn rawImg",
		"end",
		"",
		"local function applyClipboardOverlay(rawClip)",
		"\tif not ENABLE_CUSTOM_OVERLAY or type(rawClip) ~= \"string\" or rawClip == \"\" then return rawClip end",
		"\tif OVERLAY_CONFIG.CLIPBOARD[rawClip] ~= nil then return tostring(OVERLAY_CONFIG.CLIPBOARD[rawClip]) end",
		"\tif OVERLAY_CONFIG.IDS[rawClip] ~= nil then return tostring(OVERLAY_CONFIG.IDS[rawClip]) end",
		"\tif OVERLAY_CONFIG.LINKS[rawClip] ~= nil then return tostring(OVERLAY_CONFIG.LINKS[rawClip]) end",
		"\treturn rawClip",
		"end",
		"",
		"local watched = setmetatable({}, { __mode = \"k\" })",
		"local function hookGui(obj)",
		"\tif not obj or watched[obj] then return end",
		"\twatched[obj] = true",
		"\tif obj:IsA(\"TextLabel\") or obj:IsA(\"TextButton\") or obj:IsA(\"TextBox\") then",
		"\t\tif obj.Text ~= \"\" then obj.Text = applyStringOverlay(obj.Text) end",
		"\t\tobj:GetPropertyChangedSignal(\"Text\"):Connect(function()",
		"\t\t\tlocal mapped = applyStringOverlay(obj.Text)",
		"\t\t\tif mapped ~= obj.Text then obj.Text = mapped end",
		"\t\tend)",
		"\telseif obj:IsA(\"ImageLabel\") or obj:IsA(\"ImageButton\") then",
		"\t\tif obj.Image ~= \"\" then obj.Image = applyImageOverlay(obj.Image) end",
		"\t\tobj:GetPropertyChangedSignal(\"Image\"):Connect(function()",
		"\t\t\tlocal mapped = applyImageOverlay(obj.Image)",
		"\t\t\tif mapped ~= obj.Image then obj.Image = mapped end",
		"\t\tend)",
		"\tend",
		"end",
		"",
		"local function watchRoot(root)",
		"\tif not root then return end",
		"\tpcall(function()",
		"\t\tfor _, d in ipairs(root:GetDescendants()) do hookGui(d) end",
		"\t\troot.DescendantAdded:Connect(function(d) task.defer(hookGui, d) end)",
		"\tend)",
		"end",
		"",
		"if gethui then pcall(function() watchRoot(gethui()) end) end",
		"watchRoot(CoreGui)",
		"if not PlayerGui and LocalPlayer then PlayerGui = LocalPlayer:FindFirstChildOfClass(\"PlayerGui\") end",
		"if PlayerGui then watchRoot(PlayerGui) end",
		"",
		"local clipFn = setclipboard or toclipboard",
		"if hookfunction and type(clipFn) == \"function\" then",
		"\tpcall(function()",
		"\t\tlocal oldClip",
		"\t\toldClip = hookfunction(clipFn, function(txt, ...)",
		"\t\t\tif type(txt) == \"string\" then txt = applyClipboardOverlay(txt) end",
		"\t\t\treturn oldClip(txt, ...)",
		"\t\tend)",
		"\tend)",
		"end",
	}, "\n")
end

local function sanitizeFileName(name)
	local clean = tostring(name or ""):gsub("^%s+", ""):gsub("%s+$", "")
	clean = clean:gsub('[\\/:*?"<>|]', "_")
	return clean
end

local function saveToDeltaFolder(rawName)
	local writeFn = writefile or (syn and syn.write_file) or filewrite
	if not writeFn then
		return false, "Executor không hỗ trợ writefile!"
	end

	local cleanName = sanitizeFileName(rawName)
	if cleanName == "" then
		return false, "Vui lòng đặt tên file trước khi lưu!"
	end

	if not cleanName:lower():match("%.txt$") then
		cleanName = cleanName .. ".txt"
	end

	if type(makefolder) == "function" then
		pcall(function()
			if type(isfolder) == "function" then
				if not isfolder("delta") then
					makefolder("delta")
				end
			else
				makefolder("delta")
			end
		end)
	end

	local fullPath = "delta/" .. cleanName
	local content = buildRunnableOverlayScript()

	local ok, err = pcall(function()
		writeFn(fullPath, content)
	end)

	if ok then
		return true, "Đã lưu script chạy được -> " .. fullPath
	end
	return false, "Lỗi lưu file: " .. tostring(err)
end
